# RPC backend

file(GLOB MNN_RPC_SRC
    ${CMAKE_CURRENT_LIST_DIR}/*.cpp
    ${CMAKE_CURRENT_LIST_DIR}/*.hpp)

add_library(MNNRPC OBJECT ${MNN_RPC_SRC})
target_compile_definitions(MNNRPC PRIVATE MNN_RPC_ENABLED=1)
list(APPEND MNN_TARGETS MNNRPC)
list(APPEND MNN_OBJECTS_TO_LINK $<TARGET_OBJECTS:MNNRPC>)
